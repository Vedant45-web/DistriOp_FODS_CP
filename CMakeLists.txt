cmake_minimum_required(VERSION 3.16)
project(DistriOpt VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(distriopt_server src/main.cpp src/http_server.cpp src/cluster.cpp src/scheduler.cpp src/storage.cpp)
target_include_directories(distriopt_server PRIVATE include)
add_executable(distriopt_worker src/worker_agent.cpp)
target_include_directories(distriopt_worker PRIVATE include)
add_executable(distriopt_tests tests/core_tests.cpp src/cluster.cpp src/scheduler.cpp src/storage.cpp)
target_include_directories(distriopt_tests PRIVATE include)

if(WIN32)
  target_link_libraries(distriopt_server PRIVATE ws2_32)
  target_link_libraries(distriopt_worker PRIVATE ws2_32)
else()
  target_link_libraries(distriopt_server PRIVATE pthread)
  target_link_libraries(distriopt_worker PRIVATE pthread)
  target_link_libraries(distriopt_tests PRIVATE pthread)
endif()
if(MSVC)
  target_compile_options(distriopt_server PRIVATE /W4)
  target_compile_options(distriopt_worker PRIVATE /W4)
  target_compile_options(distriopt_tests PRIVATE /W4)
else()
  target_compile_options(distriopt_server PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(distriopt_worker PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(distriopt_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
